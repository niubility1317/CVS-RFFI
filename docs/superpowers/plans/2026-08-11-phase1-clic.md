CLIC_LAGS = (1, 2, 4, 8)
CLIC_INPUT_LENGTH = 256
CLIC_EMBED_DIM = 160
CLIC_INIT_SEED = 7281164
CLIC_EXTRA_PARAMETER_COUNT = 32529
FORMAL_LEO_WEAK_SCENARIOS = ("leo_clear_weak", "leo_low_elev_weak", "leo_rain_weak")
FROZEN_FOLDS = ("F1", "F2", "F3", "F4", "F5", "F6")

@dataclass(frozen=True)
class CLICConfig:
    frozen_mode: bool
    operator_mode: str
    input_length: int = CLIC_INPUT_LENGTH
    embed_dim: int = CLIC_EMBED_DIM

@dataclass
class CLICTokenBatch:
    tokens: torch.Tensor
    valid_mask: torch.Tensor
    reliability: torch.Tensor
    valid_fraction: torch.Tensor
    reliability_mean: torch.Tensor

def totalized_clic_tokens(received_i: torch.Tensor, *, operator_mode: str) -> CLICTokenBatch:
    _validate_received_i(received_i, operator_mode=operator_mode)
    batch, _, length = received_i.shape
    z = torch.complex(received_i[:, 0], received_i[:, 1])
    amplitude = torch.hypot(received_i[:, 0], received_i[:, 1])
    phase = torch.zeros_like(z)
    positive = amplitude > 0
    phase[positive] = z[positive] / amplitude[positive]
    token_blocks, masks, reliabilities = [], [], []
    for lag in CLIC_LAGS:
        left, center, right = slice(0, length - 2 * lag), slice(lag, length - lag), slice(2 * lag, length)
        valid_inner = positive[:, left] & positive[:, center] & positive[:, right]
        block = received_i.new_zeros((batch, 4, length))
        mask = torch.zeros((batch, length), dtype=torch.bool, device=received_i.device)
        reliability = received_i.new_zeros((batch, length))
        mask[:, center] = valid_inner
        a_left, a_center, a_right = amplitude[:, left], amplitude[:, center], amplitude[:, right]
        ratio = torch.minimum(torch.minimum(a_left, a_center), a_right) / torch.maximum(
            torch.maximum(a_left, a_center), a_right
        )
        reliability[:, center][valid_inner] = ratio[valid_inner]
        if operator_mode == "complex_local_invariant_curvature":
            u = phase[:, right] * phase[:, left] * phase[:, center].conj().square()
            h = torch.log(a_right[valid_inner]) + torch.log(a_left[valid_inner]) - 2 * torch.log(a_center[valid_inner])
            block[:, 0, center][valid_inner] = u.real[valid_inner]
            block[:, 1, center][valid_inner] = u.imag[valid_inner]
            block[:, 2, center][valid_inner] = h.clamp(-8, 8)
            block[:, 3, center][valid_inner] = ratio[valid_inner]
        else:
            block[:, 0, center][valid_inner] = phase[:, right].real[valid_inner]
            block[:, 1, center][valid_inner] = phase[:, right].imag[valid_inner]
            block[:, 2, center][valid_inner] = phase[:, left].real[valid_inner]
            block[:, 3, center][valid_inner] = phase[:, left].imag[valid_inner]
        token_blocks.append(block)
        masks.append(mask)
        reliabilities.append(reliability)
    tokens = torch.cat(token_blocks, dim=1)
    valid_mask = torch.stack(masks, dim=1)
    reliability = torch.stack(reliabilities, dim=1)
    _require_all_finite(tokens, reliability)
    valid_count = valid_mask.sum(dim=(1, 2))
    reliability_mean = reliability.sum(dim=(1, 2)) / valid_count.clamp_min(1)
    reliability_mean = torch.where(valid_count > 0, reliability_mean, torch.zeros_like(reliability_mean))
    return CLICTokenBatch(tokens, valid_mask, reliability, valid_mask.float().mean(dim=(1, 2)), reliability_mean)
```

Implementation must compute `p=x/|x|` only on strictly positive finite amplitudes; G channels are `[Re(u),Im(u),clip(h,-8,8),r]`, C channels are `[Re(p[t+l]),Im(p[t+l]),Re(p[t-l]),Im(p[t-l])]`, and every nonfinite intermediate raises `CLICRuntimeError`.

@dataclass
class CLICForwardResult:
    z_id: torch.Tensor
    q_clic: torch.Tensor
    token_batch: CLICTokenBatch

class CLICFusion(nn.Module):
    def __init__(self, *, embed_dim: int = 160, input_length: int = 256):
        super().__init__()
        self.depthwise = nn.Conv1d(16, 16, 5, padding=2, groups=16, bias=False)
        self.gn1 = nn.GroupNorm(4, 16)
        self.pointwise = nn.Conv1d(16, 32, 1, bias=False)
        self.gn2 = nn.GroupNorm(8, 32)
        self.embed = nn.Linear(32, 160, bias=True)
        self.correction = nn.Linear(160, 160, bias=False)
        self.gate_norm = nn.LayerNorm(320)
        self.gate = nn.Linear(320, 1, bias=True)
        initialize_clic_module_(self, seed=CLIC_INIT_SEED)

    def forward(self, received_i, z_base, *, operator_mode):
        token_batch = totalized_clic_tokens(received_i, operator_mode=operator_mode)
        channel_mask = token_batch.valid_mask.repeat_interleave(4, dim=1).to(token_batch.tokens.dtype)
        position_mask = token_batch.valid_mask.any(dim=1, keepdim=True).to(token_batch.tokens.dtype)
        encoded = F.silu(self.gn1(self.depthwise(token_batch.tokens * channel_mask)))
        encoded = F.silu(self.gn2(self.pointwise(encoded))) * position_mask
        denominator = position_mask.sum(dim=2).clamp_min(1)
        pooled = encoded.sum(dim=2) / denominator
        token_embedding = self.embed(pooled)
        gate_input = self.gate_norm(torch.cat((z_base, token_embedding), dim=1))
        gamma = token_batch.reliability_mean * torch.sigmoid(self.gate(gate_input)).squeeze(1)
        full_fallback = (token_batch.valid_mask.sum(dim=(1, 2)) == 0).to(z_base.dtype)
        gamma = gamma * (1 - full_fallback)
        z_id = z_base + gamma[:, None] * self.correction(token_embedding)
        q_clic = torch.stack(
            (gamma, token_batch.reliability_mean, token_batch.valid_fraction, full_fallback), dim=1
        )
        return CLICForwardResult(z_id=z_id, q_clic=q_clic, token_batch=token_batch)
```

Initialize encoder weights with the frozen seed, `correction.weight` as `0.01*orthogonal`, gate weight as zero, and gate bias as `logit(0.1)`. Restore caller CPU/CUDA RNG states after initialization.

def strict_clic_warm_start(model, checkpoint_state, *, checkpoint_sha256):
    current = model.state_dict()
    clic_keys = tuple(k for k in current if k.startswith("id_backbone.clic."))
    old_keys = tuple(k for k in current if k not in clic_keys)
    if set(checkpoint_state) != set(old_keys):
        raise CLICWarmStartError("checkpoint keys differ from exact pre-CLIC state")
    for key in old_keys:
        if checkpoint_state[key].shape != current[key].shape or checkpoint_state[key].dtype != current[key].dtype:
            raise CLICWarmStartError(f"checkpoint tensor contract drifted: {key}")
    old_sha_before = state_sha256({key: checkpoint_state[key] for key in old_keys})
    clic_sha_before = state_sha256({key: current[key] for key in clic_keys})
    merged = {**{key: checkpoint_state[key] for key in old_keys}, **{key: current[key] for key in clic_keys}}
    model.load_state_dict(merged, strict=True)
    loaded = model.state_dict()
    if state_sha256({key: loaded[key] for key in old_keys}) != old_sha_before:
        raise CLICWarmStartError("existing state changed during strict load")
    if state_sha256({key: loaded[key] for key in clic_keys}) != clic_sha_before:
        raise CLICWarmStartError("CLIC initialization changed during strict load")
    return {"checkpoint_sha256": checkpoint_sha256, "existing_state_sha256": old_sha_before,
